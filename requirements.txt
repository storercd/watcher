# Runtime dependencies: Watcher otherwise uses only the Python stdlib
# (tkinter, urllib, json, threading, subprocess).
#
# certifi: a plain `python3` run verifies HTTPS certs via the trust store the
# Python.org installer wires up, but a PyInstaller-frozen .app has no access
# to that (or to the macOS Keychain), so outbound HTTPS (GitHub release
# checks, ntfy.sh away notifications) fails CERTIFICATE_VERIFY_FAILED unless
# we hand `ssl` an explicit CA bundle (see watcher/core/net.py).
certifi>=2024.0.0

# pyobjc-framework-Cocoa is optional and macOS-only: it powers the bouncing
# Dock icon in watcher/notifiers/macos.py. It's imported lazily and the
# notifier degrades gracefully (banner + sound only, no bounce) if it's
# missing, so it's not a hard requirement.
pyobjc-framework-Cocoa>=10.0; sys_platform == "darwin"

# Dev/test dependencies:
pytest>=7.0
ruff>=0.16

# Build dependency: used by scripts/build_macos_app.sh to package Watcher as
# a standalone .app bundle (so it can be pinned to the macOS Dock with its
# own icon instead of a generic Python rocket).
pyinstaller>=6.0; sys_platform == "darwin"
